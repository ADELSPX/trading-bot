# 1. نشرة أخبار الصباح — كل 9 صباحاً
hermes cron create \
  --name "نشرة أخبار الصباح" \
  --schedule "0 9 * * *" \
  --skill daily-tech-news \
  --prompt "Fetch today's tech news from Techmeme via Jina AI: curl -sL --max-time 15 \"https://r.jina.ai/https://techmeme.com\" | head -300. Summarize top 5 stories in Arabic, bullet format. Include source links." \
  --deliver origin \
  --toolsets terminal,web
